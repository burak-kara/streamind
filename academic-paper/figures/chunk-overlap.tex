% F3 — Chunking and novel extraction timing.
% Three chunks of 5 s, sliding by 4 s. Each chunk after the first carries
% 1 s of overlap (orange) with the previous chunk; the novel extractor
% emits only the remaining 4 s (green).
%
% Coordinate system: x in seconds (via x=\unitW), y in millimeters
% (via y=1mm). Node `minimum width/height` need explicit mm values, so
% width-in-mm is pre-computed via pgfmath from the dimensionless `*Sec`
% durations and the per-second width.
\def\chunkSec{5}            % seconds per chunk
\def\overlapSec{1}          % seconds of overlap with previous
\def\novelSec{4}            % chunkSec - overlapSec
\def\advanceSec{4}          % seconds slide between chunks
\def\timeMaxSec{14}         % time axis end
\def\tickEverySec{2}        % tick spacing
\def\unitWmm{5.5}           % horizontal mm per second
\def\unitW{\unitWmm mm}     % matching dimension for x scaling

\def\barHmm{4}              % chunk bar height (mm)
\def\rowGapMm{2}            % vertical gap between stacked chunks (mm)
\def\axisYOffMm{5}          % gap between axis line and first chunk (mm)

\pgfmathsetmacro{\chunkWmm}{\chunkSec*\unitWmm}
\pgfmathsetmacro{\overlapWmm}{\overlapSec*\unitWmm}
\pgfmathsetmacro{\novelWmm}{\novelSec*\unitWmm}
\pgfmathsetmacro{\rowPitchMm}{\barHmm + \rowGapMm}

\begin{tikzpicture}[
    x=\unitW, y=1mm,
    every node/.style={font=\scriptsize, inner sep=0pt},
    novel/.style={
        draw=colorGreen!75, fill=colorGreen!25,
        rounded corners=1pt,
        minimum height=\barHmm mm, anchor=south west,
        text=colorGreen!40!black
    },
    overlap/.style={
        draw=colorOrange!85, fill=colorOrange!35,
        rounded corners=1pt,
        minimum height=\barHmm mm, anchor=south west,
        text=colorOrange!55!black
    },
    axisline/.style={-{Latex[length=1.2mm]}, color=labelColor, semithick},
    tickstyle/.style={color=labelColor, semithick}
]
    % Time axis. x-units are seconds, y-units are mm.
    \draw[axisline] (0, 0) -- (\timeMaxSec + 0.5, 0)
        node[right, font=\tiny, text=labelColor]{time (s)};
    \foreach \t in {0, \tickEverySec, ..., \timeMaxSec}{
        \draw[tickstyle] (\t, 0) -- (\t, -1.2)
            node[below, font=\tiny, text=labelColor]{\t};
    }

    % Bottom-left anchors for each chunk, chained via calc.
    \coordinate (b0) at (0, \axisYOffMm);
    \coordinate (b1) at ($(b0) + (\advanceSec, \rowPitchMm)$);
    \coordinate (b2) at ($(b1) + (\advanceSec, \rowPitchMm)$);

    % Chunk 0: pure novel (no preceding chunk to overlap with).
    \node[novel, minimum width=\chunkWmm mm] at (b0) (c0n) {chunk 0};

    % Chunks 1 and 2: 1 s overlap segment + 4 s novel segment.
    \node[overlap, minimum width=\overlapWmm mm] at (b1) (c1o) {};
    \node[novel,   minimum width=\novelWmm mm,
          anchor=south west] at (c1o.south east) (c1n) {chunk 1};

    \node[overlap, minimum width=\overlapWmm mm] at (b2) (c2o) {};
    \node[novel,   minimum width=\novelWmm mm,
          anchor=south west] at (c2o.south east) (c2n) {chunk 2};

    % Legend row above the top chunk, chained relatively.
    \node[novel, minimum width=3mm, minimum height=3mm,
          above=4mm of c2o.north west, anchor=south west, xshift=-8mm] (legN) {};
    \node[anchor=west, right=1mm of legN] (legNlbl) {novel (emitted)};
    \node[overlap, minimum width=3mm, minimum height=3mm,
          right=4mm of legNlbl.east, anchor=west] (legO) {};
    \node[anchor=west, right=1mm of legO] {overlap (discarded)};
\end{tikzpicture}
